let tongzi = 'ExportRule'
var Content = GetContent()
let url = bucketGet("ExportUrl", "url")

function main() {
    if (Content == '规则管理') {
        selectBucket()
    } else {
        exportCk()
    }
}

function exportCk() {
    let rule = bucketKeys(tongzi) || []
    let data = []
    rule.forEach(item => {
        data.push({
            Bucket: item,
            rule: JSON.parse(bucketGet(tongzi, item))
        });
    })

    let choose = []

    if (!Content) {
        choose = data;
    } else {
        chooseRule(rule)
        var msg = ShuRu()

        if (msg === "0") {
            choose = data;
        } else if (msg) {
            msg = msg.split(',');
            choose = msg.map(item => data[parseInt(item) - 1])
        } else {
            return
        }
    }

    Debug(JSON.stringify(choose))

    choose.forEach(RuleConfig => {
        UpdateCk(RuleConfig)
    })
}

function QueryCk(RuleConfig) {
    let str = []
    let wangchao = bucketKeys(RuleConfig.Bucket)
    Debug(JSON.stringify(wangchao))

    wangchao.forEach(item => {
        let info = bucketGet(RuleConfig.Bucket, item)
        Debug(info)
        info = JSON.parse(info)
        info.forEach(item => {
            let data = ''
            RuleConfig.rule.keys.forEach((config, index) => {
                Debug(config, index)
                data += `${item[config]}${RuleConfig.rule.separators[index] ? RuleConfig.rule.separators[index] : ''}`
            })
            str.push(data)
        })
    })
    sendChunks(str)
}

function UpdateCk(RuleConfig) {
    let str = []
    let wangchao = bucketKeys(RuleConfig.Bucket)
    Debug(JSON.stringify(wangchao))

    wangchao.forEach(item => {
        let info = bucketGet(RuleConfig.Bucket, item)
        Debug(info)
        info = JSON.parse(info)
        info.forEach(item => {
            let data = ''
            RuleConfig.rule.keys.forEach((config, index) => {
                Debug(config, index)
                let value = item[config]
                if (config === 'authorization_time') {
                    value = value.replace(/-/g, '/');
                }
                data += `${value}${RuleConfig.rule.separators[index] ? RuleConfig.rule.separators[index] : ''}`
            })
            str.push({
                filename: `${item.cookie}.text`,
                data
            })
        })
    })
    UpdateFile(RuleConfig.rule.directory, str)
}

function UpdateFile(directory, filesData) {

    let row = {
        directory,
        filesData
    }

    Debug(JSON.stringify(row))
    request({
        "url": url, //请求链接
        "method": "post", //请求方法
        body: row,
        "timeOut": 30000,
    }, function (error, response, header, body) {
        Debug(body)
        let res = JSON.parse(body)
        if (res.code === 200) {
            if (!Content) {
                notifyMasters(res.message)
            } else {
                sendText(res.message)
            }
        } else {
            if (!Content) {
                notifyMasters(JSON.stringify(res))
            } else {
                sendText(JSON.stringify(res))
            }
        }
    })
}

function chooseRule(rule) {
    let str = []
    rule.forEach((key, index) => {
        let data = bucketGet(tongzi, key)
        str.push(`${index + 1}：桶名：${key}\n    导出格式：${ConversionRules(data)}`)
    })
    str.unshift(`请选择要导出的规则：\n0：全部`)
    str.push(`\n选择其中几个请用英文逗号,隔开，如：1,3,5`)
    sendChunks(str)
}

function selectBucket() {
    while (true) {
        let rule = bucketKeys(tongzi) || []

        printRule(rule)
        var msg = ShuRu()
        if (msg === '0') {
            addRule(rule)
            // break
        } else if (msg < 0) {
            try {
                bucketDel(tongzi, rule[Math.abs(msg) - 1])
                sendText('删除成功')
                // break
            } catch (err) {
                sendText("输入有误，请重新输入")
            }
        } else if (msg === false) {
            break
        }
    }
}

function addRule(rule) {
    let SingleRule = {
        Bucket: "",
        rule: new RuleFormat([], [])
    };

    sendText("请输入要提取的桶名");
    let bucketName = RuleShuRu();
    if (bucketName === false) return; // 如果用户退出或超时，直接结束函数
    SingleRule.Bucket = bucketName;

    sendText("请依次输入键名和分隔符，输入q结束。");
    sendText("请输入一个键名，例如：mobile");
    // sendText("例如：输入 'mobile'（回车）'&'（回车）'pwd'（回车）'#'（回车）'remark'（回车）'q'");

    let input;
    let isKey = true;
    while (true) {
        input = RuleShuRu();
        if (input === false || input === 'q' || input === 'Q') break; // 用户退出或输入q/Q

        if (isKey) {
            SingleRule.rule.keys.push(input);
            isKey = false;
            sendText("当前规则: " + SingleRule.rule.toString());
            sendText("请输入分隔符，例如：-（或输入q结束）");
        } else {
            SingleRule.rule.separators.push(input);
            isKey = true;
            sendText("当前规则: " + SingleRule.rule.toString());
            sendText("请输入下一个键名,例如：phone");
        }
    }

    // 只有当最后输入的是分隔符时，才移除最后一个分隔符
    if (!isKey && SingleRule.rule.separators.length > SingleRule.rule.keys.length - 1) {
        SingleRule.rule.separators.pop();
    }

    if (SingleRule.rule.keys.length > 0) {
        rule.push(SingleRule.Bucket);
        // bucketSet(tongzi, SingleRule.Bucket, JSON.stringify(SingleRule.rule));
        // sendText("添加成功");
        sendText("最终规则：" + SingleRule.rule.toString());

        sendText("请输入文件保存路径，例：/home/look/" + SingleRule.Bucket)
        var msg = ShuRu()
        if (msg) {
            SingleRule.rule.directory = msg
            bucketSet(tongzi, SingleRule.Bucket, JSON.stringify(SingleRule.rule));
            sendText("添加成功");
        } else {
            sendText("未输入文件保存路径，取消本次配置。")
        }
    } else {
        sendText("未添加任何规则");
    }
}

class RuleFormat {
    constructor(keys, separators) {
        this.keys = keys;
        this.separators = separators;
    }

    toString() {
        return this.keys.reduce((result, key, index) => {
            if (index === 0) return key;
            return result + (this.separators[index - 1] || '') + key;
        }, '');
    }
}

function ConversionRules(obj) {
    obj = JSON.parse(obj)
    return obj.keys.reduce((result, key, index) => {
        if (index === 0) return key;
        return result + (obj.separators[index - 1] || '') + key;
    }, '');
}

function printRule(rule) {
    let str = ["请选择导出规则进行编辑：\n(-数字删除,0添加,q退出)"]
    rule.forEach((key, index) => {
        let data = bucketGet(tongzi, key)
        str.push(`${index + 1}.桶名：${key}\n导出格式：${ConversionRules(data)}`)
    })
    sendChunks(str)
}

function RuleShuRu() {
    var msg = input(60000, 6000)
    if (msg == null) {
        sendText("超时，60秒内未回复，取消本次配置。")
        return false
    } else if (msg == "q" || msg == "Q") {
        // sendText("已退出会话");
        return false
    } else {
        return msg;
    }
}

function ShuRu() {
    var msg = input(60000, 6000)
    if (msg === null) {
        sendText("超时，60秒内未回复，取消本次配置。")
        return false
    } else if (msg == "q" || msg == "Q") {
        sendText("已退出会话");
        return false
    } else {
        return msg;
    }
}

function splitArrayIntoChunks(array, chunkSize) {
    let results = [];
    for (let i = 0; i < array.length; i += chunkSize) {
        results.push(array.slice(i, i + chunkSize));
    }
    return results;
}


function sendChunks(array) {
    let chunks = splitArrayIntoChunks(array, 100)
    chunks.forEach(item => {
        sendText(item.join('\n'))
        sleep(1000)
    })
}
